01function quickSort(arr, low, high) {
02  if (low < high) {
03    const pivotIndex = partition(arr, low, high);
04    quickSort(arr, low, pivotIndex - 1);
05    quickSort(arr, pivotIndex + 1, high);
06  }
07}

09function partition(arr, low, high) {
10  const pivot = arr[high];
11  let i = low - 1;
12  for (let j = low; j < high; j++) {
13    if (arr[j] < pivot) {
14      i++;
15      swap(arr, i, j);
16    }
17  }
18  swap(arr, i + 1, high);
19  return i + 1;
20}
